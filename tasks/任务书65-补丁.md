#### 1.1 主属性按「职业+专精」动态联动

**新增常量映射表**（js/app.js 或独立常量模块）：

```js
const CLASS_SPEC_PRIMARY_STAT = {
  '死亡骑士': { '*': '力量' },
  '恶魔猎手': { '浩劫': '敏捷', '复仇': '敏捷', '噬灭': '智力' },
  '德鲁伊':   { '平衡': '智力', '恢复': '智力', '野性': '敏捷', '守护': '敏捷' },
  '唤魔师':   { '*': '智力' },
  '猎人':     { '*': '敏捷' },
  '法师':     { '*': '智力' },
  '武僧':     { '织雾': '智力', '*': '敏捷' },
  '圣骑士':   { '神圣': '智力', '*': '力量' },
  '牧师':     { '*': '智力' },
  '潜行者':   { '*': '敏捷' },
  '萨满':     { '元素': '智力', '恢复': '智力', '*': '敏捷' },
  '术士':     { '*': '智力' },
  '战士':     { '*': '力量' },
};

function getPrimaryStat(className, specName) {
  const classMap = CLASS_SPEC_PRIMARY_STAT[className];
  if (!classMap) return '力量'; // 兜底
  return classMap[specName] || classMap['*'] || '力量';
}
```

**交互规则**：
- 套装模式下，选择职业 → 主属性不变（等专精）
- 选择专精后 → 主属性自动更新（如圣骑士·神圣 → 智力；圣骑士·惩戒 → 力量）
- 主属性下拉框只读或自动填充，不允许用户手动修改（减少误操作）

**新增字段说明**：

| 字段 | 来源 | 获取方式 |
|---|---|---|
| `primaryStat` | 插件硬编码（同 WP1 映射表） | 根据 className + specName 查表 |
| `secondaryStats` | Tooltip 扫描 | 解析装备 Tooltip 中的绿字属性 |
| `tokenBossId` / `tokenBossName` | 兑换物关联 | 通过兑换物 itemID 查其掉落首领 |

#### 2.2 副属性扫描（Tooltip 解析）

```lua
local function ScanSecondaryStats(itemID)
  local stats = {}
  local tooltip = CreateFrame("GameTooltip", "WoWButlerScanTip", nil, "GameTooltipTemplate")
  tooltip:SetOwner(UIParent, "ANCHOR_NONE")
  tooltip:SetItemByID(itemID)
  
  for i = 1, tooltip:NumLines() do
    local text = _G["WoWButlerScanTipTextLeft" .. i]:GetText()
    if text then
      -- 匹配 "+58 爆击" "+143 急速" 等格式
      local value, statName = text:match("%+(%d+)%s+([^%s]+)")
      if value and statName then
        -- 过滤有效副属性
        if statName:match("爆击") or statName:match("急速") or 
           statName:match("精通") or statName:match("全能") then
          table.insert(stats, statName)
        end
      end
    end
  end
  
  tooltip:Hide()
  return stats
end
```

**注意**：Tooltip 扫描可能因客户端语言（简中/繁中/英文）返回不同文本。建议只取**两条**副属性（装备固定两条），存储为数组。

#### 2.3 兑换物→首领映射

**获取逻辑**：

```lua
-- 套装兑换物（token）与首领的关联
```

**推荐方案**：插件端内置一个**兑换物→首领的预置映射表**（因为兑换物 itemID 是固定的，20 个兑换物左右）。

**问题**：兑换物 itemID 需要事先知道。如果你不知道，插件可以先导出「兑换物 itemID」，你手动查一下 wowhead 补映射表，再更新插件。

**简化方案**：插件导出时，套装装备的 `tokenBossName` 先留空，你手动在网站导入页面补填。
